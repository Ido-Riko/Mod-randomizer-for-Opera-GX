(async () => {
  const sleep = ms => new Promise(r => setTimeout(r, ms));

  // ===================================================================================
  // HELPER: Extract mod data from an <a> tag
  // ===================================================================================
  function parseModCard(a) {
    const href = a.getAttribute("href");
    if (!href || href.endsWith("/mods/")) return null;

    const titleEl = a.querySelector(".gx-header-xs-alt, h3, .mod-card-title");
    let title = titleEl?.textContent?.trim();

    if (!title) {
      const ariaLabel = a.getAttribute("aria-label");
      title = (ariaLabel && ariaLabel !== "Mod card") ? ariaLabel : a.textContent?.trim();
    }

    if (title) {
      title = title.replace(/Show me\s*:?\s*/gi, "").replace(/\s+/g, " ").trim();
    }

    if (!title) return null;

    const fullUrl = href.startsWith("http") ? href : `https://store.gx.me${href}`;
    return { title, url: fullUrl };
  }

  // ===================================================================================
  // PHASE 1: Scrape Live DOM
  // ===================================================================================
  async function scrapeLiveDomMods(scrolls = 30, delay = 3500) {
    const catalog = {};
    const seenUrls = new Set();
    let lastModCount = 0;
    let stableCount = 0;

    console.log("🎯 Incrementally scrolling and capturing mods from the live DOM...");

    for (let i = 0; i < scrolls; i++) {
      window.scrollBy({ top: 800, behavior: 'smooth' });
      await sleep(delay);

      document.querySelectorAll("a[href*='/mods/']").forEach(a => {
        const mod = parseModCard(a);
        if (!mod || seenUrls.has(mod.url)) return;

        seenUrls.add(mod.url);
        catalog[mod.title] = { url: mod.url, source: "live-dom" };
      });

      const currentModCount = seenUrls.size;

      if (currentModCount === lastModCount) {
        stableCount++;
        console.log(`  ⚠️ Scroll ${i + 1}/${scrolls}: No new mods observed (${stableCount}/3)`);
        if (stableCount >= 3) {
          console.log("  ✅ End of visible dynamic stream reached.");
          break;
        }
      } else {
        stableCount = 0;
        const diff = currentModCount - lastModCount;
        console.log(`  ✅ Scroll ${i + 1}/${scrolls}: +${diff} new mods. (Live unique: ${currentModCount})`);
      }

      lastModCount = currentModCount;
    }

    console.log(`\n✅ Finished Phase 1: Captured ${seenUrls.size} unique mods.`);
    console.log("------------------------------------------------------------\n");
    return { catalog, seenUrls };
  }

  // ===================================================================================
  // PHASE 2: Scrape All Mods via Pagination
  // ===================================================================================
  async function scrapeAllModsViaPagination(initialData) {
    const catalog = { ...initialData.catalog };
    const seenUrls = new Set(initialData.seenUrls);

    let page = 1;
    let consecutiveEmptyPages = 0;
    let lastLoggedCount = Object.keys(catalog).length;

    console.log("📄 Fetching mods via pagination (store.gx.me/mods?page=N)...");

    while (true) {
      try {
        const response = await fetch(`https://store.gx.me/mods?page=${page}&tagAlias=desktop`);
        if (!response.ok) throw new Error(`HTTP ${response.status}`);

        const text = await response.text();
        const doc = new DOMParser().parseFromString(text, "text/html");

        // Collect all valid mod cards rendered in this fetched page
        const modLinks = Array.from(doc.querySelectorAll('a[href*="/mods/"]'))
          .filter(a => {
            const h = a.getAttribute("href");
            return h && !h.endsWith("/mods/");
          });

        // Genuine page termination: server returned zero mod cards
        if (modLinks.length === 0) {
          console.log(`  ℹ️ Page ${page}: No mod cards found in markup.`);
          consecutiveEmptyPages++;
        } else {
          // The page has content: reset the empty page counter regardless of duplicates
          consecutiveEmptyPages = 0;

          let newFromPage = 0;
          modLinks.forEach(link => {
            const mod = parseModCard(link);
            if (!mod || seenUrls.has(mod.url)) return;

            seenUrls.add(mod.url);
            catalog[mod.title] = { url: mod.url, source: "pagination" };
            newFromPage++;
          });

          if (page % 5 === 0 || newFromPage > 0) {
            console.log(`  - Page ${page}: ${modLinks.length} items on page (+${newFromPage} new, total: ${seenUrls.size})`);
          }
        }

        const totalMods = seenUrls.size;
        if (totalMods - lastLoggedCount >= 500) {
          const milestone = Math.floor(totalMods / 500) * 500;
          console.log(`  ⭐ Milestone Reached: Over ${milestone} mods indexed!`);
          lastLoggedCount = milestone;
        }

      } catch (err) {
        console.error(`  ❌ Failed to fetch page ${page}:`, err.message);
        consecutiveEmptyPages++;
      }

      // Only terminate when the store genuinely runs out of pages
      if (consecutiveEmptyPages >= 3) {
        console.log(`  ✅ Reached end of pagination (3 consecutive empty responses).`);
        break;
      }

      page++;
      await sleep(250);
    }

    console.log(`\n✅ Finished Phase 2: Total unique mods: ${seenUrls.size}.`);
    return catalog;
  }

  // ===================================================================================
  // EXECUTION & DIRECT EXPORT
  // ===================================================================================
  console.log("🚀 Initializing GX Mods Scraper...");
  console.log("============================================================");
  const startTime = Date.now();

  const phase1Result = await scrapeLiveDomMods();
  const allMods = await scrapeAllModsViaPagination(phase1Result);

  // Normalize to pure { "Mod Title": { "url": "..." } }
  const formattedOutput = {};
  for (const title in allMods) {
    formattedOutput[title] = { url: allMods[title].url };
  }

  // Alphabetical sort
  const sortedMods = Object.fromEntries(
    Object.entries(formattedOutput).sort(([a], [b]) => a.localeCompare(b))
  );

  const elapsedSeconds = ((Date.now() - startTime) / 1000).toFixed(1);
  const totalCount = Object.keys(sortedMods).length;

  console.log("\n============================================================");
  console.log("🎉 SCRAPING COMPLETE! 🎉");
  console.log("============================================================");
  console.log(`⏱️  Elapsed Time: ${elapsedSeconds}s`);
  console.log(`📊 Live DOM Captured: ${phase1Result.seenUrls.size}`);
  console.log(`📊 Total Catalog: ${totalCount}`);
  console.log("============================================================\n");

  console.log("💾 Generating download for 'url-catalog.json'...");
  const jsonOutput = JSON.stringify(sortedMods, null, 2);
  const blob = new Blob([jsonOutput], { type: 'application/json' });
  const downloadUrl = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = downloadUrl;
  a.download = 'url-catalog.json';
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(downloadUrl);
  console.log("✅ Download triggered successfully.");

  return sortedMods;
})();
